# TO DO LIST
---
TABLE task.text AS "待办事项", task.due AS "截止日期"
FROM "你的任务文件夹"
WHERE task.completed = false AND task.due = date(today)
SORT task.due ASC
# DeadLine
---
TABLE task.text AS "待办事项", task.due AS "截止日期"
FROM "你的任务文件夹"
WHERE task.completed = false AND task.due >= date(thismonth) AND task.due <= date(endofmonth)
SORT task.due ASC
# Done List
---
TABLE task.text AS "已完成任务", task.completed AS "完成时间"
FROM "你的任务文件夹"
WHERE task.completed = true AND task.completed >= date(today)
SORT task.completed DESC
