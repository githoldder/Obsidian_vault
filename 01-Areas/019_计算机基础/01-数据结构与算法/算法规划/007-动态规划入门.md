## 一、DP核心思想

**一句话理解**：把大问题拆成子问题，子问题不重复计算（记忆化）

**三要素**：
1. **状态定义**：`dp[i]` 或 `dp[i][j]` 表示什么
2. **状态转移**：如何从已知状态推导出未知状态
3. **初始化**：边界条件是什么

## 二、经典DP模型

### 2.1 斐波那契/爬楼梯

```cpp
// 状态：dp[i] = 爬到第i阶的方法数
// 转移：dp[i] = dp[i-1] + dp[i-2]
int climbStairs(int n) {
    if (n <= 2) return n;
    int dp[n + 1];
    dp[1] = 1; dp[2] = 2;
    for (int i = 3; i <= n; i++)
        dp[i] = dp[i-1] + dp[i-2];
    return dp[n];
}

// 空间优化
int climbStairs(int n) {
    if (n <= 2) return n;
    int prev2 = 1, prev1 = 2;
    for (int i = 3; i <= n; i++) {
        int curr = prev1 + prev2;
        prev2 = prev1;
        prev1 = curr;
    }
    return prev1;
}
```

### 2.2 背包问题

#### 0/1背包

```cpp
// 状态：dp[i][j] = 前i个物品，容量j时的最大价值
// 转移：dp[i][j] = max(dp[i-1][j], dp[i-1][j-w[i]] + v[i])
int knapsack01(vector<int>& w, vector<int>& v, int W) {
    int n = w.size();
    vector<vector<int>> dp(n + 1, vector<int>(W + 1));
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j <= W; j++) {
            dp[i][j] = dp[i-1][j];  // 不选第i个
            if (j >= w[i-1])
                dp[i][j] = max(dp[i][j], dp[i-1][j-w[i-1]] + v[i-1]);
        }
    }
    return dp[n][W];
}

// 空间优化（倒序遍历）
int knapsack01(vector<int>& w, vector<int>& v, int W) {
    vector<int> dp(W + 1);
    for (int i = 0; i < w.size(); i++) {
        for (int j = W; j >= w[i]; j--)  // 倒序！
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
    }
    return dp[W];
}
```

#### 完全背包

```cpp
// 与01背包的区别：物品可以选无限次
// 正序遍历
int completeKnapsack(vector<int>& w, vector<int>& v, int W) {
    vector<int> dp(W + 1);
    for (int i = 0; i < w.size(); i++) {
        for (int j = w[i]; j <= W; j++)  // 正序！
            dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
    }
    return dp[W];
}
```

**记忆口诀**：
- 0/1背包：倒序（每个物品只能用一次）
- 完全背包：正序（可以重复用）

### 2.3 最大子数组和（Kadane算法）

```cpp
// 状态：dp[i] = 以第i个元素结尾的最大子数组和
// 转移：dp[i] = max(nums[i], dp[i-1] + nums[i])
int maxSubArray(vector<int>& nums) {
    int n = nums.size();
    int dp = nums[0];
    int res = dp;
    for (int i = 1; i < n; i++) {
        dp = max(nums[i], dp + nums[i]);
        res = max(res, dp);
    }
    return res;
}
```

### 2.4 最长递增子序列（LIS）

```cpp
// O(n²) 版本
int lengthOfLIS(vector<int>& nums) {
    int n = nums.size();
    vector<int> dp(n, 1);  // dp[i] = 以nums[i]结尾的LIS长度
    int res = 1;
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (nums[j] < nums[i])
                dp[i] = max(dp[i], dp[j] + 1);
        }
        res = max(res, dp[i]);
    }
    return res;
}

// O(nlogn) 版本（二分优化）
int lengthOfLIS(vector<int>& nums) {
    vector<int> tails;  // tails[i] = 长度为i+1的递增子序列的最小末尾
    for (int x : nums) {
        auto it = lower_bound(tails.begin(), tails.end(), x);
        if (it == tails.end()) tails.push_back(x);
        else *it = x;
    }
    return tails.size();
}
```

### 2.5 编辑距离

```cpp
// 状态：dp[i][j] = word1[0..i-1]转成word2[0..j-1]的最小操作数
int minDistance(string word1, string word2) {
    int m = word1.size(), n = word2.size();
    vector<vector<int>> dp(m + 1, vector<int>(n + 1));
    
    for (int i = 0; i <= m; i++) dp[i][0] = i;  // 删除
    for (int j = 0; j <= n; j++) dp[0][j] = j;  // 插入
    
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (word1[i-1] == word2[j-1])
                dp[i][j] = dp[i-1][j-1];
            else
                dp[i][j] = 1 + min({dp[i-1][j],      // 删除
                                    dp[i][j-1],      // 插入
                                    dp[i-1][j-1]});  // 替换
        }
    }
    return dp[m][n];
}
```
