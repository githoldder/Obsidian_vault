const int MAXN = 1005;

## 二、图的遍历

### DFS（深度优先）

```cpp
vector<bool> vis;

void dfs(int u) {
    vis[u] = true;
    for (auto& [v, w] : g[u]) {
        if (!vis[v]) dfs(v);
    }
}

// 非递归版本（用栈）
void dfs_iter(int start) {
    stack<int> st;
    st.push(start);
    while (!st.empty()) {
        int u = st.top(); st.pop();
        if (vis[u]) continue;
        vis[u] = true;
        for (auto& [v, w] : g[u]) {
            if (!vis[v]) st.push(v);
        }
    }
}
```

### BFS（广度优先）

```cpp
void bfs(int start) {
    queue<int> q;
    q.push(start);
    vis[start] = true;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (auto& [v, w] : g[u]) {
            if (!vis[v]) {
                vis[v] = true;
                q.push(v);
            }
        }
    }
}
```

## 三、最短路径

### Dijkstra（单源最短路，无负权边）

```cpp
vector<int> dijkstra(int start) {
    int n = g.size();
    vector<int> dist(n, INF);
    vector<bool> vis(n);
    dist[start] = 0;
    
    // 优先队列：{距离, 结点}
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<>> pq;
    pq.push({0, start});
    
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (vis[u]) continue;
        vis[u] = true;
        
        for (auto& [v, w] : g[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}
```

**时间复杂度**：O((n+m)logn)

### Floyd（多源最短路）

```cpp
void floyd(vector<vector<int>>& dist) {
    int n = dist.size();
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (dist[i][k] + dist[k][j] < dist[i][j])
                    dist[i][j] = dist[i][k] + dist[k][j];
}
```

**时间复杂度**：O(n³)，适合n≤500

## 四、最小生成树

### Kruskal算法（并查集）

```cpp
int kruskal(vector<tuple<int, int, int>>& edges, int n) {
    // edges: {w, u, v}
    sort(edges.begin(), edges.end());
    UnionFind uf(n);
    int mst = 0, cnt = 0;
    
    for (auto& [w, u, v] : edges) {
        if (uf.unite(u, v)) {
            mst += w;
            cnt++;
        }
    }
    return cnt == n - 1 ? mst : -1;  // -1表示不连通
}
```

### Prim算法（类似Dijkstra）

```cpp
int prim(vector<vector<pair<int, int>>>& g, int start) {
    int n = g.size();
    vector<int> dist(n, INF);
    vector<bool> vis(n);
    dist[start] = 0;
    
    int mst = 0;
    for (int i = 0; i < n; i++) {
        int u = -1;
        for (int j = 0; j < n; j++)
            if (!vis[j] && (u == -1 || dist[j] < dist[u]))
                u = j;
        
        if (dist[u] == INF) return -1;  // 不连通
        vis[u] = true;
        mst += dist[u];
        
        for (auto& [v, w] : g[u])
            if (!vis[v] && w < dist[v])
                dist[v] = w;
    }
    return mst;
}
```

## 五、拓扑排序

```cpp
vector<int> topoSort(vector<vector<int>>& g, vector<int>& indegree) {
    int n = g.size();
    queue<int> q;
    for (int i = 0; i < n; i++)
        if (indegree[i] == 0) q.push(i);
    
    vector<int> res;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        res.push_back(u);
        for (int v : g[u]) {
            if (--indegree[v] == 0) q.push(v);
        }
    }
    return res.size() == n ? res : vector<int>();  // 空表示有环
}
```
