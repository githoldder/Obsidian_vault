## 三、Supabase 后端服务

### 3.1 数据库设计 (PostgreSQL)

**schema.sql** 核心表结构：

```sql
-- 用户表 (与 Supabase Auth 集成)
CREATE TABLE profiles (
    id UUID REFERENCES auth.users ON DELETE CASCADE PRIMARY KEY,
    nickname VARCHAR(50) DEFAULT '考研er',
    exam_type VARCHAR(10) DEFAULT '英语一',
    target_year INTEGER DEFAULT 2026,
    avatar_url TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 文章表 (95篇真题)
CREATE TABLE articles (
    id SERIAL PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    exam_year INTEGER NOT NULL,
    exam_type VARCHAR(10) NOT NULL,
    category VARCHAR(20),
    full_text TEXT NOT NULL,
    translation TEXT,
    difficulty VARCHAR(10) DEFAULT '中等',
    word_count INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 句子表
CREATE TABLE sentences (
    id SERIAL PRIMARY KEY,
    article_id INTEGER REFERENCES articles(id) ON DELETE CASCADE,
    sentence_order INTEGER NOT NULL,
    content TEXT NOT NULL,
    translation TEXT,
    keywords JSONB DEFAULT '[]',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 单词表
CREATE TABLE words (
    id SERIAL PRIMARY KEY,
    word VARCHAR(100) NOT NULL UNIQUE,
    phonetic_uk VARCHAR(100),
    phonetic_us VARCHAR(100),
    meaning_zh TEXT NOT NULL,
    difficulty VARCHAR(10) DEFAULT '中等',
    frequency INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 用户背词记录表 (核心表)
CREATE TABLE user_words (
    id SERIAL PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE,
    word_id INTEGER REFERENCES words(id) ON DELETE CASCADE,
    source_article_id INTEGER REFERENCES articles(id) ON DELETE SET NULL,
    source_sentence_id INTEGER REFERENCES sentences(id) ON DELETE SET NULL,
    status VARCHAR(20) DEFAULT 'new',
    show_count INTEGER DEFAULT 0,
    mastery_level INTEGER DEFAULT 0,
    mastered BOOLEAN DEFAULT FALSE,
    learned_at TIMESTAMP WITH TIME ZONE,
    last_reviewed_at TIMESTAMP WITH TIME ZONE,
    next_review_at TIMESTAMP WITH TIME ZONE,
    spelling_correct_count INTEGER DEFAULT 0,
    spelling_error_count INTEGER DEFAULT 0,
    is_spelling_error BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(user_id, word_id)
);

-- 语法知识点表
CREATE TABLE grammar_points (
    id SERIAL PRIMARY KEY,
    point_name VARCHAR(100) NOT NULL,
    category VARCHAR(50) NOT NULL,
    sub_category VARCHAR(50),
    content_md TEXT NOT NULL,
    examples JSONB DEFAULT '[]',
    keywords JSONB DEFAULT '[]',
    difficulty VARCHAR(10) DEFAULT '中等',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 句子-语法关联表
CREATE TABLE sentence_grammar (
    id SERIAL PRIMARY KEY,
    sentence_id INTEGER REFERENCES sentences(id) ON DELETE CASCADE,
    grammar_point_id INTEGER REFERENCES grammar_points(id) ON DELETE CASCADE,
    match_type VARCHAR(20) DEFAULT 'keyword',
    match_score REAL DEFAULT 0.8,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(sentence_id, grammar_point_id)
);

-- 用户提醒设置表
CREATE TABLE reminders (
    id SERIAL PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE,
    slot VARCHAR(20) NOT NULL CHECK (slot IN ('morning', 'noon', 'night')),
    time VARCHAR(10) NOT NULL DEFAULT '07:30',
    enabled BOOLEAN DEFAULT TRUE,
    snooze_count INTEGER DEFAULT 0,
    last_snoozed_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(user_id, slot)
);

-- 学习记录表 (用于看板统计)
CREATE TABLE study_logs (
    id SERIAL PRIMARY KEY,
    user_id UUID REFERENCES auth.users ON DELETE CASCADE,
    log_type VARCHAR(20) NOT NULL CHECK (log_type IN ('word', 'review', 'reading', 'grammar')),
    item_id INTEGER,
    action VARCHAR(50),
    duration_seconds INTEGER DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 创建索引优化查询
CREATE INDEX idx_articles_year ON articles(exam_year);
CREATE INDEX idx_articles_type ON articles(exam_type);
CREATE INDEX idx_sentences_article ON sentences(article_id);
CREATE INDEX idx_words_word ON words(word);
CREATE INDEX idx_user_words_user ON user_words(user_id);
CREATE INDEX idx_user_words_next_review ON user_words(next_review_at);
CREATE INDEX idx_user_words_status ON user_words(status);
CREATE INDEX idx_study_logs_user_date ON study_logs(user_id, created_at);
```

### 3.2 Supabase 配置

**环境变量 (.env.local)**：

```bash
# Supabase 配置
VITE_SUPABASE_URL=https://your-project.supabase.co
VITE_SUPABASE_ANON_KEY=your-anon-key

# Gemini API 配置
VITE_GEMINI_API_KEY=your-gemini-api-key

# 应用配置
VITE_APP_NAME=考验英语
VITE_APP_VERSION=1.0.0
```

**Supabase 客户端初始化 (src/api.ts)**：

```typescript
import { createClient } from '@supabase/supabase-js'

const supabaseUrl = import.meta.env.VITE_SUPABASE_URL
const supabaseAnonKey = import.meta.env.VITE_SUPABASE_ANON_KEY

export const supabase = createClient(supabaseUrl, supabaseAnonKey)

// API 封装示例
export const api = {
  // 用户相关
  auth: {
    signUp: (email: string, password: string) =>
      supabase.auth.signUp({ email, password }),
    signIn: (email: string, password: string) =>
      supabase.auth.signInWithPassword({ email, password }),
    signOut: () => supabase.auth.signOut(),
    getUser: () => supabase.auth.getUser(),
  },

  // 单词相关
  words: {
    getAll: () => supabase.from('words').select('*'),
    getById: (id: number) => supabase.from('words').select('*').eq('id', id).single(),
    search: (query: string) =>
      supabase.from('words').select('*').ilike('word', `%${query}%`),
  },

  // 用户背词记录
  userWords: {
    getQueue: (userId: string) =>
      supabase.from('user_words')
        .select('*, words(*)')
        .eq('user_id', userId)
        .lte('next_review_at', new Date().toISOString())
        .order('next_review_at'),

    addWord: (userId: string, wordId: number, articleId?: number) =>
      supabase.from('user_words').insert({
        user_id: userId,
        word_id: wordId,
        source_article_id: articleId,
        learned_at: new Date().toISOString(),
        next_review_at: new Date(Date.now() + 10 * 60000).toISOString(), // 10分钟后
      }),

    updateReview: (id: number, data: any) =>
      supabase.from('user_words').update(data).eq('id', id),
  },

  // 文章相关
  articles: {
    getAll: () => supabase.from('articles').select('*').order('exam_year', { ascending: false }),
    getById: (id: number) =>
      supabase.from('articles').select('*, sentences(*)').eq('id', id).single(),
  },

  // 看板统计
  dashboard: {
    getStats: async (userId: string) => {
      // 使用 Supabase RPC 调用存储过程
      const { data, error } = await supabase
        .rpc('get_user_stats', { user_id: userId })
      return { data, error }
    },
  },
}
```

### 3.3 Supabase Edge Functions (可选)

对于复杂业务逻辑，使用 Supabase Edge Functions：

```typescript
// supabase/functions/ai-chat/index.ts
import { serve } from 'https://deno.land/std@0.168.0/http/server.ts'
import { GoogleGenerativeAI } from 'https://esm.sh/@google/generative-ai'

serve(async (req) => {
  const { message, context } = await req.json()

  const genAI = new GoogleGenerativeAI(Deno.env.get('GEMINI_API_KEY'))
  const model = genAI.getGenerativeModel({ model: 'gemini-pro' })

  // RAG: 从 Supabase 检索相关知识
  const { data: relevantDocs } = await supabase
    .from('grammar_points')
    .select('*')
    .textSearch('content_md', message)

  const prompt = `
    基于以下考研英语知识库内容回答问题：
    ${relevantDocs?.map(d => d.content_md).join('\n')}

    用户问题：${message}
  `

  const result = await model.generateContent(prompt)
  const response = result.response.text()

  return new Response(JSON.stringify({ answer: response }), {
    headers: { 'Content-Type': 'application/json' },
  })
})
```

## 四、AI 集成 (Gemini API)

### 4.1 RAG 知识库问答

```typescript
// src/services/ai.ts
import { GoogleGenerativeAI } from '@google/generative-ai'

const genAI = new GoogleGenerativeAI(import.meta.env.VITE_GEMINI_API_KEY)

export const aiService = {
  // 单词/语法答疑
  async askQuestion(question: string, context?: any) {
    const model = genAI.getGenerativeModel({ model: 'gemini-pro' })

    const prompt = `
      你是考研英语学习助手，请基于真题知识回答问题。
      ${context ? `上下文：${JSON.stringify(context)}` : ''}

      问题：${question}

      要求：
      1. 答案紧扣考研范围
      2. 提供真题例句说明
      3. 简洁明了，适合备考场景
    `

    const result = await model.generateContent(prompt)
    return result.response.text()
  },

  // 作文纠错
  async correctEssay(essay: string) {
    const model = genAI.getGenerativeModel({ model: 'gemini-pro' })

    const prompt = `
      请对以下考研英语作文进行纠错：
      ${essay}

      只纠正拼写和语法错误，不提供模板。
    `

    const result = await model.generateContent(prompt)
    return result.response.text()
  },

  // 智能推荐
  async getRecommendations(userStats: any) {
    // 基于用户学习数据生成个性化建议
  }
}
```

| 检查项 | 接口/功能 | 预期结果 |

### 6.2 Antigravity 集成

**Antigravity 作用**：边缘计算 + AI 推理优化

```typescript
// 使用 Antigravity 优化 AI 调用
import { antigravity } from '@antigravity/sdk'

// 缓存常用 AI 回答
const cachedAnswer = await antigravity.cache.get(`qa:${questionHash}`)
if (cachedAnswer) {
  return cachedAnswer
}

// 调用 Gemini API
const answer = await aiService.askQuestion(question)

// 缓存结果
await antigravity.cache.set(`qa:${questionHash}`, answer, { ttl: 3600 })
```
