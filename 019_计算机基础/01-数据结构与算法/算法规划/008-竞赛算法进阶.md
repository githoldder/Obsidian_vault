### 经典题型

#### 区间调度问题

```cpp
// 最多能参加多少个不冲突的会议
int eraseOverlapIntervals(vector<vector<int>>& intervals) {
    // 按结束时间排序
    sort(intervals.begin(), intervals.end(), 
         [](auto& a, auto& b) { return a[1] < b[1]; });
    
    int cnt = 1, end = intervals[0][1];
    for (int i = 1; i < intervals.size(); i++) {
        if (intervals[i][0] >= end) {  // 可以参加
            cnt++;
            end = intervals[i][1];
        }
    }
    return intervals.size() - cnt;  // 需要移除的数量
}
```

// 能否到达最后一个位置
bool canJump(vector<int>& nums) {
    int maxReach = 0;
    for (int i = 0; i < nums.size(); i++) {
        if (i > maxReach) return false;
        maxReach = max(maxReach, i + nums[i]);
    }
    return true;
}

## 二、搜索进阶

### 记忆化搜索（DFS + DP）

```cpp
// 单词拆分：能否用字典中的词拼出s
unordered_map<string, bool> memo;

bool wordBreak(string s, vector<string>& wordDict) {
    unordered_set<string> dict(wordDict.begin(), wordDict.end());
    return dfs(s, dict);
}

bool dfs(string s, unordered_set<string>& dict) {
    if (s.empty()) return true;
    if (memo.count(s)) return memo[s];
    
    for (int i = 1; i <= s.size(); i++) {
        string prefix = s.substr(0, i);
        if (dict.count(prefix) && dfs(s.substr(i), dict))
            return memo[s] = true;
    }
    return memo[s] = false;
}
```

### 双向BFS

```cpp
// 单词接龙：从beginWord到endWord的最短转换序列
int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
    unordered_set<string> dict(wordList.begin(), wordList.end());
    if (!dict.count(endWord)) return 0;
    
    unordered_set<string> beginSet{beginWord}, endSet{endWord};
    int len = 1;
    
    while (!beginSet.empty() && !endSet.empty()) {
        if (beginSet.size() > endSet.size()) swap(beginSet, endSet);
        unordered_set<string> nextSet;
        
        for (string word : beginSet) {
            for (int i = 0; i < word.size(); i++) {
                char old = word[i];
                for (char c = 'a'; c <= 'z'; c++) {
                    word[i] = c;
                    if (endSet.count(word)) return len + 1;
                    if (dict.count(word)) {
                        nextSet.insert(word);
                        dict.erase(word);
                    }
                }
                word[i] = old;
            }
        }
        beginSet = nextSet;
        len++;
    }
    return 0;
}
```

## 三、数学基础

### 快速幂

```cpp
// 计算 a^n mod mod
long long fastPow(long long a, long long n, long long mod) {
    long long res = 1;
    a %= mod;
    while (n > 0) {
        if (n & 1) res = res * a % mod;  // n是奇数
        a = a * a % mod;
        n >>= 1;  // n /= 2
    }
    return res;
}
```

### GCD和LCM

```cpp
// 最大公约数
int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}

// 最小公倍数
int lcm(int a, int b) {
    return a / gcd(a, b) * b;
}
```

### 乘法逆元

```cpp
// 费马小定理：a^(p-1) ≡ 1 (mod p)，p是质数
// a的逆元 = a^(p-2) mod p
long long inv(long long a, long long p) {
    return fastPow(a, p - 2, p);
}
```

## 四、字符串算法

### KMP算法

```cpp
// 计算next数组（最长相同前后缀）
vector<int> computeNext(string& pattern) {
    int n = pattern.size();
    vector<int> next(n);
    next[0] = -1;
    int j = -1;
    for (int i = 1; i < n; i++) {
        while (j >= 0 && pattern[i] != pattern[j + 1])
            j = next[j];
        if (pattern[i] == pattern[j + 1]) j++;
        next[i] = j;
    }
    return next;
}

// KMP匹配
int kmp(string& text, string& pattern) {
    vector<int> next = computeNext(pattern);
    int j = -1;
    for (int i = 0; i < text.size(); i++) {
        while (j >= 0 && text[i] != pattern[j + 1])
            j = next[j];
        if (text[i] == pattern[j + 1]) j++;
        if (j == pattern.size() - 1) return i - j;  // 匹配成功
    }
    return -1;
}
```

## 五、高级数据结构

### 线段树（区间查询/修改）

```cpp
class SegmentTree {
    vector<int> tree, lazy;
    int n;
    
    void pushUp(int node) {
        tree[node] = tree[node * 2] + tree[node * 2 + 1];
    }
    
    void pushDown(int node, int l, int r) {
        if (lazy[node]) {
            int mid = (l + r) / 2;
            tree[node * 2] += lazy[node] * (mid - l + 1);
            tree[node * 2 + 1] += lazy[node] * (r - mid);
            lazy[node * 2] += lazy[node];
            lazy[node * 2 + 1] += lazy[node];
            lazy[node] = 0;
        }
    }
    
public:
    SegmentTree(int size) {
        n = size;
        tree.resize(4 * n);
        lazy.resize(4 * n);
    }
    
    void build(vector<int>& nums, int node, int l, int r) {
        if (l == r) { tree[node] = nums[l]; return; }
        int mid = (l + r) / 2;
        build(nums, node * 2, l, mid);
        build(nums, node * 2 + 1, mid + 1, r);
        pushUp(node);
    }
    
    void update(int node, int l, int r, int L, int R, int val) {
        if (L <= l && r <= R) {
            tree[node] += val * (r - l + 1);
            lazy[node] += val;
            return;
        }
        pushDown(node, l, r);
        int mid = (l + r) / 2;
        if (L <= mid) update(node * 2, l, mid, L, R, val);
        if (R > mid) update(node * 2 + 1, mid + 1, r, L, R, val);
        pushUp(node);
    }
    
    int query(int node, int l, int r, int L, int R) {
        if (L <= l && r <= R) return tree[node];
        pushDown(node, l, r);
        int mid = (l + r) / 2, res = 0;
        if (L <= mid) res += query(node * 2, l, mid, L, R);
        if (R > mid) res += query(node * 2 + 1, mid + 1, r, L, R);
        return res;
    }
};
```
