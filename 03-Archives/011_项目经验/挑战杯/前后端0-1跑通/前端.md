## 6. 前端参考（交由搭档）

- Vue.js组件（参考）：
    
    ```vue
    <template>
      <div>
        <input v-model="text" placeholder="输入文本" />
        <button @click="generate">生成</button>
        <p>{{ sentence }}</p>
        <audio :src="audioUrl" controls></audio>
        <div id="spectrogram"></div>
      </div>
    </template>
    
    <script>
    import Plotly from 'plotly.js-dist';
    
    export default {
      data() { return { text: '', sentence: '', audioUrl: '', spectrogramData: null }; },
      methods: {
        async generate() {
          const response = await fetch('/generate-audio-spectrogram', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ text: this.text })
          });
          const data = await response.json();
          this.sentence = data.sentence;
          this.audioUrl = data.audio_url;
          this.spectrogramData = JSON.parse(data.spectrogram);
          Plotly.newPlot('spectrogram', this.spectrogramData.data, this.spectrogramData.layout);
        }
      }
    };
    </script>
    ```
